cmake_minimum_required(VERSION 3.18)
project(RealtimeInputPipelineCore VERSION 0.1.0 LANGUAGES C CXX)

option(RIP_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(PipelineCore STATIC Sources/PipelineCore/PipelineCore.cpp)
target_include_directories(PipelineCore PUBLIC Sources/PipelineCore/include)
target_compile_features(PipelineCore PUBLIC cxx_std_17)
set_target_properties(PipelineCore PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(PipelineCore PRIVATE /W4 /WX)
else()
  target_compile_options(PipelineCore PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(RIP_ENABLE_SANITIZERS)
  if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "RIP_ENABLE_SANITIZERS requires GCC or Clang with address/undefined sanitizer support")
  endif()
  target_compile_options(PipelineCore PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(PipelineCore PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(PipelineCoreTests CppTests/core_tests.cpp)
  target_link_libraries(PipelineCoreTests PRIVATE PipelineCore)
  if(NOT MSVC)
    target_compile_options(PipelineCoreTests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME PipelineCoreTests COMMAND PipelineCoreTests)
  add_executable(PipelineCoreCHeaderProbe CppTests/c_header_probe.c)
  set_target_properties(PipelineCoreCHeaderProbe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
  target_link_libraries(PipelineCoreCHeaderProbe PRIVATE PipelineCore)
  add_test(NAME PipelineCoreCHeaderProbe COMMAND PipelineCoreCHeaderProbe)
endif()
